cmake_minimum_required(VERSION 3.25)
project(seedvr2_ncnn_precision_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(SEEDVR2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SEEDVR2_NCNN_PREFIX "${SEEDVR2_ROOT}/.deps/ncnn-install" CACHE PATH "Prepared ncnn to probe")
file(READ "${SEEDVR2_NCNN_PREFIX}/seedvr2-ncnn-commit.txt" PROBE_NCNN_COMMIT)
string(STRIP "${PROBE_NCNN_COMMIT}" PROBE_NCNN_COMMIT)
file(SHA256 "${SEEDVR2_NCNN_PREFIX}/lib/libncnn.a" PROBE_NCNN_LIBRARY_SHA256)
find_package(ncnn CONFIG REQUIRED PATHS "${SEEDVR2_NCNN_PREFIX}/lib/cmake/ncnn" NO_DEFAULT_PATH)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(NOT NCNN_VULKAN)
    message(FATAL_ERROR "The precision probe requires Vulkan")
endif()
add_executable(ncnn-precision-probe main.cpp)
target_include_directories(ncnn-precision-probe PRIVATE "${SEEDVR2_ROOT}/include" "${SEEDVR2_ROOT}/src/engine/ncnn")
target_include_directories(ncnn-precision-probe SYSTEM PRIVATE "${SEEDVR2_ROOT}/third_party")
target_compile_definitions(ncnn-precision-probe PRIVATE PROBE_NCNN_COMMIT="${PROBE_NCNN_COMMIT}" PROBE_NCNN_LIBRARY_SHA256="${PROBE_NCNN_LIBRARY_SHA256}")
target_link_libraries(ncnn-precision-probe PRIVATE ncnn OpenSSL::Crypto)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(ncnn-precision-probe PRIVATE "-Wl,--exclude-libs,ALL")
endif()
